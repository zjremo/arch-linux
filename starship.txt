starship.toml: 放置在~/.config/starship.toml

# 为 IDE 提供 schema 实现自动补全
"$schema" = 'https://starship.rs/config-schema.json'

# 在提示符之间插入空行
add_newline = true

# 将提示符中的 '❯' 替换为 '$'
[character] # 此组件名称为 'character'
success_symbol = '[\$](bold green)'

# 禁用 'package' 组件，将其隐藏
[package]
disabled = true

.zshrc:

# 可选：启用一些 Zsh 选项
setopt HIST_IGNORE_DUPS

# 加载 zsh-autosuggestions
source /usr/share/zsh/plugins/zsh-autosuggestions/zsh-autosuggestions.zsh

# 加载 zsh-syntax-highlighting（必须放在 autosuggestions 之后，且在 starship 之前！）
source /usr/share/zsh/plugins/zsh-syntax-highlighting/zsh-syntax-highlighting.zsh

# 初始化补全系统
autoload -Uz compinit
compinit

# 自定义插件样式
ZSH_AUTOSUGGEST_HIGHLIGHT_STYLE="fg=#888888"
ZSH_HIGHLIGHT_HIGHLIGHTERS=(main brackets pattern cursor)

# 设置光标颜色为白色
echo -ne "\e]12;#ffffff\a"

# clear设置alias
alias c="clear"

# lazygit设置lg
alias lg="lazygit"

# JAVA路径配置
export JAVA_HOME=/usr/lib/jvm/java-21-openjdk
export PATH=$JAVA_HOME/bin:$PATH

# Mason配置, 系统路径加入，可以访问到mason安装的lsp
export MASON="$HOME/.local/share/nvim/mason"
export PATH="$MASON/bin:$PATH"

# npm全局安装目录设置
export PATH=~/.npm-global/bin:$PATH

# claude code env config
export ANTHROPIC_AUTH_TOKEN=
export ANTHROPIC_BASE_URL=https://open.bigmodel.cn/api/anthropic
export API_TIMEOUT_MS=3000000
export CLAUDE_CODE_DISABLE_NONESSENTIAL_TRAFFIC=1
export ANTHROPIC_MODEL=glm-4.6
export ANTHROPIC_SMALL_FAST_MODEL=glm-4.5-airx

# 初始化 Starship（必须放在最后）
eval "$(starship init zsh)"
